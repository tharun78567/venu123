<script>
function generatePlan() {

    let age = document.getElementById("age").value;
    let weight = document.getElementById("weight").value;
    let height = document.getElementById("height").value;
    let goal = document.getElementById("goal").value;

    if (!age || !weight || !height) {
        alert("Please enter all details");
        return;
    }

    let plan = "";

    if (goal === "weightloss") {
        plan = `
            <h2>🔥 Weight Loss Plan</h2>
            <ul>
                <li>Monday: Walking + Squats</li>
                <li>Tuesday: Cardio</li>
                <li>Wednesday: Rest</li>
                <li>Thursday: Jogging + Push-ups</li>
                <li>Friday: Full Body Workout</li>
                <li>Saturday: Walking</li>
                <li>Sunday: Rest</li>
            </ul>
        `;
    }

    else if (goal === "muscle") {
        plan = `
            <h2>💪 Muscle Gain Plan</h2>
            <ul>
                <li>Monday: Chest + Triceps</li>
                <li>Tuesday: Back + Biceps</li>
                <li>Wednesday: Rest</li>
                <li>Thursday: Legs</li>
                <li>Friday: Shoulders + Arms</li>
                <li>Saturday: Full Body</li>
                <li>Sunday: Rest</li>
            </ul>
        `;
    }

    else {
        plan = `
            <h2>🏃 General Fitness Plan</h2>
            <ul>
                <li>Monday: Full Body Workout</li>
                <li>Tuesday: Walking</li>
                <li>Wednesday: Cardio</li>
                <li>Thursday: Rest</li>
                <li>Friday: Full Body Workout</li>
                <li>Saturday: Walking + Stretching</li>
                <li>Sunday: Rest</li>
            </ul>
        `;
    }

    document.getElementById("result").innerHTML = plan;
}
</script>
